% SLIDE 2 — Introduzione
\begin{frame}{Introduzione}
    \begin{itemize}
        \item Il modello SIR descrive la diffusione di un'infezione in una popolazione chiusa.
        \item L'evoluzione futura dipende \textbf{solo dallo stato presente} $\rightarrow$ proprietà di Markov.
        \item Lo spazio degli stati è \textbf{finito} $\rightarrow$ catena di Markov a tempo discreto.
        \item Il problema non è epidemiologico: è un pretesto per applicare il formalismo.
    \end{itemize}
    \vspace{0.5cm}
    \textbf{Riferimento}: Capitolo 7 del corso — Catene di Markov a tempo discreto.
    
    \note{
        Perché studiare un'epidemia con le catene di Markov?
        Il motivo è semplice: il modello SIR si presta naturalmente a essere
        descritto come un processo markoviano. La popolazione è divisa in tre
        gruppi: Suscettibili, Infetti e Rimossi. L'evoluzione futura dipende
        solo dallo stato presente, non dalla storia passata. Questa è
        esattamente la proprietà di Markov. Inoltre lo spazio degli stati
        è finito, quindi siamo nel caso classico delle catene a tempo discreto
        con spazio finito, che è esattamente l'argomento del capitolo 7.
    }
\end{frame}

% SLIDE 4 — Richiamo: Catene di Markov
\begin{frame}{Catene di Markov (Cap. 7)}
    \textbf{Definizione}: Processo stocastico $\{X_t\}$ a tempo discreto con la \textbf{proprietà di Markov}:
    \[ \mathbb{P}(X_{t+1} | X_t, X_{t-1}, \dots, X_0) = \mathbb{P}(X_{t+1} | X_t) \]
    
    \textbf{Matrice di transizione $P$}: $P(i,j) =$ probabilità di andare da $i$ a $j$ in un passo.
    \begin{itemize}
        \item $P$ è \textbf{stocastica}: $\sum_j P(i,j) = 1$ per ogni $i$.
    \end{itemize}
    
    \vspace{0.3cm}
    \textbf{Stato assorbente}: $P(i,i) = 1$.\\
    Una volta entrato, non si esce più.

    \note{
        Richiamiamo brevemente i concetti fondamentali delle catene di Markov,
        così come li abbiamo studiati nel capitolo 7.
        Una catena di Markov a tempo discreto è un processo stocastico
        {X_t} con la proprietà che la distribuzione
        dello stato futuro dipende solo dallo stato presente e non da
        quelli passati. Questa si chiama proprietà di Markov.
        La catena è descritta da una matrice di transizione P, dove
        l'elemento P(i,j) è la probabilità di passare dallo stato i
        allo stato j in un passo.
        La matrice P è stocastica: ogni riga contiene probabilità non
        negative che sommano a 1.
        Un concetto fondamentale per questo progetto è quello di
        stato assorbente. Uno stato i si dice assorbente se, una volta
        raggiunto, non lo si abbandona mai più.
    }
\end{frame}

% SLIDE 5 — Modello SIR
\begin{frame}{Modello SIR}
    Popolazione chiusa di \textbf{$N$ individui}. Tre compartimenti:
    
    \begin{center}
    \begin{tabular}{lll}
    \toprule
    Simbolo & Nome & Descrizione \\
    \midrule
    $S_t$ & Suscettibili & Possono essere infettati \\
    $I_t$ & Infetti & Possono trasmettere l'infezione \\
    $R_t$ & Rimossi & Guariti, immuni, non infettivi \\
    \bottomrule
    \end{tabular}
    \end{center}
    
    \vspace{0.3cm}
    \textbf{Vincolo}: $S_t + I_t + R_t = N$ per ogni $t$.\\
    \textbf{Stato}: $X_t = (S_t, I_t, R_t)$.

    \note{
        Veniamo al modello SIR. Consideriamo una popolazione chiusa di
        dimensione costante N. In ogni istante discreto t, la popolazione
        è suddivisa in tre compartimenti:
        S_t (Suscettibili), I_t (Infetti) e R_t (Rimossi).
        Vale il vincolo di conservazione: S_t + I_t + R_t = N per ogni t.
        Questo perché la popolazione è chiusa: non ci sono nascite, morti
        o migrazioni.
        Lo stato del sistema al tempo t è il vettore X_t = (S_t, I_t, R_t).
        La dinamica è governata da due meccanismi casuali indipendenti:
        il contagio e la guarigione.
    }
\end{frame}

% SLIDE 6 — Spazio degli stati
\begin{frame}{Spazio degli stati}
    \textbf{Definizione}: $E = \{(s, i, r) \in \mathbb{N}_0^3 : s + i + r = N\}$
    
    \textbf{Cardinalità}: $|E| = \binom{N+2}{2} = \frac{(N+1)(N+2)}{2}$
    
    \textbf{Esempi}:
    \begin{itemize}
        \item $N = 3 \Rightarrow |E| = 10$ stati
        \item $N = 100 \Rightarrow |E| = 5151$ stati
    \end{itemize}
    
    Lo spazio è \textbf{finito} $\rightarrow$ siamo nel caso classico delle catene a spazio finito (Cap. 7).\\
    Per $N$ piccolo la matrice $P$ si calcola esplicitamente. Per $N$ grande si usa la simulazione Monte Carlo.

    \note{
        Lo spazio degli stati della catena è l'insieme di tutte le terne
        (s, i, r) di numeri interi non negativi la cui somma è N.
        Quanti sono gli stati? Il numero di modi di distribuire N oggetti
        indistinguibili in 3 scatole è dato dalla formula scritta sulla slide.
        Per N = 100, abbiamo 5151 stati. È uno spazio
        finito, come richiesto dalla teoria delle catene di Markov, ma
        troppo grande per calcolare esplicitamente la matrice di transizione.
        Attenzione: il fatto che lo spazio sia grande non è un problema
        concettuale. La teoria delle catene di Markov vale per qualsiasi
        spazio finito.
    }
\end{frame}

% SLIDE 7 — Probabilità di transizione
\begin{frame}{Probabilità di transizione}
    \textbf{Contagio}: $C_t \sim \text{Binomiale}(S_t, \beta \cdot I_t / N)$
    
    \textbf{Guarigione}: $G_t \sim \text{Binomiale}(I_t, \gamma)$
    
    \textbf{Aggiornamento}:
    \begin{align*}
        S_{t+1} &= S_t - C_t \\
        I_{t+1} &= I_t + C_t - G_t \\
        R_{t+1} &= R_t + G_t
    \end{align*}
    
    \textbf{Parametri}:
    \begin{itemize}
        \item $\beta \in [0,1]$ : tasso di trasmissione
        \item $\gamma \in [0,1]$ : tasso di guarigione
    \end{itemize}
    
    Contagio e guarigione sono \textbf{indipendenti}. Probabilità composta: prodotto delle due Binomiali.

    \note{
        Come evolve il sistema da un passo al successivo? Abbiamo due
        fenomeni casuali indipendenti.
        Primo: il contagio. La probabilità che un singolo Suscettibile diventi infetto
        in un passo è β·I_t/N. Quindi il numero di nuovi contagi C_t segue una Binomiale.
        Secondo: la guarigione. Ogni Infetto guarisce con probabilità γ. 
        Il numero di guarigioni G_t segue anch'esso una Binomiale.
        L'aggiornamento dello stato è semplice aritmetica.
        Notate che i due meccanismi sono indipendenti. La probabilità
        di passare da uno stato all'altro è il prodotto delle probabilità
        Binomiali.
    }
\end{frame}

% SLIDE 8 — Esempio N=3
\begin{frame}{Esempio: N=3, $\beta=0.5$, $\gamma=0.3$}
    Stato iniziale: \textbf{(2, 1, 0)}
    
    \begin{center}
    \begin{tabular}{cccc}
    \toprule
    $C$ (contagi) & $G$ (guarigioni) & Stato finale & Probabilità \\
    \midrule
    0 & 0 & (2, 1, 0) & 0.486 \\
    0 & 1 & (2, 0, 1) & 0.208 \\
    1 & 0 & (1, 2, 0) & 0.194 \\
    1 & 1 & (1, 1, 1) & 0.083 \\
    \bottomrule
    \end{tabular}
    \end{center}
    
    \vspace{0.3cm}
    $\sum \text{probabilità} = 1$ \checkmark — riga stocastica

    \note{
        Facciamo un esempio concreto con N=3. Prendiamo β = 0.5 e γ = 0.3. 
        Partiamo dallo stato (2, 1, 0): cioè 2 Suscettibili, 1 Infetto, 0 Rimossi.
        La probabilità di contagio per ogni Suscettibile è 0.5 · 1/3.
        Quindi il numero di contagi C può essere 0 o 1.
        La probabilità che nessuno si infetti è circa 0.694.
        La probabilità che un Suscettibile si infetti è circa 0.278.
        Per la guarigione: l'unico Infetto guarisce con prob. 0.3, non guarisce con 0.7.
        Combinando i due eventi indipendenti, otteniamo 4 casi possibili,
        le cui probabilità sommano a 1.
        La riga della matrice di transizione corrispondente a questo stato è
        completamente determinata.
    }
\end{frame}

% SLIDE 9 — Matrice di transizione
\begin{frame}{Matrice di transizione $P$}
    \textbf{Elemento generico}:
    \[ P((s,i,r), (s',i',r')) = \sum_c \sum_g \text{Bin}(s, c, \beta i/N) \cdot \text{Bin}(i, g, \gamma) \cdot \mathbf{1}_{\{s' = s-c, i' = i+c-g, r' = r+g\}} \]
    
    \textbf{Proprietà}:
    \begin{itemize}
        \item $P$ è \textbf{stocastica}: righe sommano a 1.
        \item Per $N=3$: $P$ è $10 \times 10$, calcolabile esplicitamente.
        \item Per $N=100$: $5151$ stati $\rightarrow$ $\sim 26$ milioni di elementi $\rightarrow$ simulazione.
    \end{itemize}
    
    \vspace{0.3cm}
    \textit{La matrice esiste sempre, anche quando non la scriviamo per intero.}

    \note{
        La matrice di transizione P è una matrice quadrata di dimensione
        |E| × |E|. Per N=3 abbiamo 10 stati. La matrice P è quindi 10 × 10. 
        Un elemento generico si calcola come scritto sulla slide, dove la
        funzione indicatrice seleziona solo le combinazioni
        (c, g) che portano allo stato desiderato.
        Per N=100 avremmo 5151 stati e la matrice
        avrebbe circa 26 milioni di elementi — possibile in teoria ma
        dispendioso in pratica.
    }
\end{frame}

% SLIDE 10 — Classificazione stati
\begin{frame}{Classificazione degli stati}
    \textbf{Stato $(s, i, r)$ con $i = 0$} $\rightarrow$ \textbf{assorbente}
    \begin{itemize}
        \item $P((s,0,r), (s,0,r)) = 1$
        \item Non ci sono infetti: nessun contagio, nessuna guarigione.
    \end{itemize}
    
    \textbf{Stato $(s, i, r)$ con $i > 0$} $\rightarrow$ \textbf{transitorio}
    \begin{itemize}
        \item Da ogni stato con $I>0$ si raggiunge un assorbente.
        \item Non si può tornare a $I>0$ dopo l'assorbimento.
    \end{itemize}
    
    \vspace{0.3cm}
    \textbf{Conseguenza}: l'estinzione dell'infezione è \textbf{certa}: $\mathbb{P}(\tau < \infty) = 1$.\\
    \textit{La catena è una catena assorbente (Cap. 7).}

    \note{
        Classifichiamo ora gli stati della catena. Se i = 0, il tasso di contagio 
        è zero. Quindi C = 0. Anche le guarigioni sono 0
        perché non ci sono infetti. Quindi lo stato non cambia mai:
        P((s, 0, r), (s, 0, r)) = 1.
        Questo significa che tutti gli stati con i = 0 sono assorbenti.
        Tutti gli stati con i > 0 sono transitori: da essi si può
        raggiungere uno stato assorbente in un numero finito di passi,
        e non si può tornare indietro.
        La catena è quindi assorbente: l'assorbimento in
        uno stato con I=0 avviene con probabilità 1. Questo è coerente
        con l'idea che l'epidemia prima o poi si estingue.
    }
\end{frame}

% SLIDE 11 — Algoritmo simulazione
\begin{frame}{Algoritmo di simulazione}
    \begin{enumerate}
        \item Inizializza: $S = N - I_0$, $I = I_0$, $R = 0$
        \item Loop $t = 0, 1, \dots, T_{\max}$:
        \begin{enumerate}
            \item Se $I = 0 \rightarrow$ fermati
            \item $C \sim \text{Binomiale}(S, \beta \cdot I/N)$
            \item $G \sim \text{Binomiale}(I, \gamma)$
            \item $S \leftarrow S - C$, $I \leftarrow I + C - G$, $R \leftarrow R + G$
            \item Se $I = 0 \rightarrow$ fermati
        \end{enumerate}
        \item Restituisci traiettoria $(S, I, R)$
    \end{enumerate}
    
    \textbf{Parametri}: $N=100$, $\beta=0.2$, $\gamma=0.1$, $I_0=5$, $T_{\max}=200$, $M=1000$\\
    \textbf{$R_0 = \beta/\gamma = 2.0$} — ogni infetto genera 2 nuovi infetti in media.

    \note{
        Passiamo ora alla simulazione. L'algoritmo per generare una
        traiettoria è riportato sulla slide.
        Ripetendo questo procedimento molte volte (diciamo M = 1000),
        otteniamo un campione di traiettorie. Su questo campione possiamo
        calcolare medie, deviazioni standard, distribuzioni empiriche.
        Con i parametri scelti, il numero riproduttivo di base è
        R₀ = β/γ = 2. Questo significa che, nelle fasi iniziali,
        ogni infetto genera in media 2 nuovi infetti.
    }
\end{frame}

% SLIDE 12 — Codice Python
\begin{frame}[fragile]{Codice Python — Passo della catena}
\begin{lstlisting}[language=Python]
import numpy as np

N, BETA, GAMMA = 100, 0.2, 0.1
I0, T_MAX, M = 5, 200, 1000

def next_state(s, i, r, n=N, beta=BETA, gamma=GAMMA):
    if i == 0:
        return s, 0, r
    p_si = beta * i / n
    new_infections = np.random.binomial(s, p_si)
    recoveries = np.random.binomial(i, gamma)
    return s - new_infections, i + new_infections - recoveries, r + recoveries
\end{lstlisting}

    \note{
        Ecco il codice Python che implementa quanto descritto. È volutamente
        essenziale: la funzione next\_state esegue un singolo passo della
        catena, sfruttando np.random.binomial per generare le transizioni.
        Il codice riflette esattamente la struttura della catena di Markov:
        next\_state è l'analogo di un passo della catena.
    }
\end{frame}

        Terzo: il numero medio di infetti al picco è 29 su 100.

        dell'estinzione. Questo è coerente con R₀ = 2.
    }
\end{frame}

% SLIDE 18 — Teoria vs Simulazione
\begin{frame}{Teoria vs Simulazione}
    \begin{center}
    \begin{tabular}{ll}
    \toprule
    Previsione teorica & Verifica simulativa \\
    \midrule
    Assorbimento certo ($\mathbb{P}=1$) & \checkmark 1000/1000 casi \\
    $\tau = $ soluzione di $(I-Q)t = 1$ & \checkmark $\tau$ medio $\approx 68$ (stimato) \\
    $R_\infty$ è una variabile casuale & \checkmark Distribuzione empirica ottenuta \\
    \bottomrule
    \end{tabular}
    \end{center}
    
    \vspace{0.5cm}
    \textbf{Punto chiave}: La simulazione NON sostituisce la teoria. La simulazione \textbf{stima} quantità che la teoria definisce. È la teoria che dà significato ai numeri.

    \note{
        Confrontiamo ora i risultati della simulazione con le previsioni
        teoriche. La teoria delle catene assorbenti ci dice che l'assorbimento
        avviene con probabilità 1 (confermato).
        Il tempo medio di assorbimento si ricaverebbe dalla matrice Q, 
        ma per N=100 calcolarla esplicitamente è proibitivo. 
        La simulazione ci fornisce una stima: tau medio = 68.
        La matrice di transizione esiste anche se non la
        calcoliamo esplicitamente. La simulazione Monte Carlo è un modo
        di campionare da questa matrice senza doverla costruire per intero.
    }
\end{frame}

% SLIDE 19 — Limiti
\begin{frame}{Limiti del modello}
    \begin{center}
    \begin{tabular}{ll}
    \toprule
    Limite & Impatto \\
    \midrule
    Popolazione chiusa & Ignora dinamiche demografiche \\
    $\beta, \gamma$ costanti & Ignora stagionalità e interventi \\
    Mescolamento omogeneo & Ignora reti sociali \\
    Nessuna reinfezione & Non sempre realistico \\
    \bottomrule
    \end{tabular}
    \end{center}
    
    \vspace{0.5cm}
    \textbf{Ma}: questi limiti sono \textbf{voluti}. Il focus è sulla struttura markoviana, non sull'accuratezza epidemiologica.

    \note{
        È importante essere onesti sui limiti del modello.
        Primo: la popolazione è chiusa.
        Secondo: i parametri β e γ sono costanti.
        Terzo: il mescolamento è omogeneo. Ogni Suscettibile ha la stessa probabilità di incontrare ogni Infetto.
        Quarto: non c'è reinfezione.
        Tutti questi limiti sono accettabili perché il nostro obiettivo
        non è fare previsioni epidemiologiche, ma dimostrare l'applicazione
        del formalismo delle catene di Markov. Anzi, la semplicità del
        modello aiuta a mettere in luce la struttura markoviana.
    }
\end{frame}

% SLIDE 20 — Sviluppi futuri
\begin{frame}{Sviluppi futuri}
    \begin{enumerate}
        \item \textbf{Modello SEIR}: aggiungere compartimento Esposti.
        \item \textbf{Parametri tempo-varianti}: $\beta(t), \gamma(t)$ per simulare misure di contenimento.
        \item \textbf{Confronto con ODE}: limite deterministico per $N \to \infty$.
        \item \textbf{Matrice P per N piccolo}: heatmap della matrice di transizione.
    \end{enumerate}
    
    \vspace{0.5cm}
    Tutte le estensioni mantengono la \textbf{struttura di catena di Markov}.

    \note{
        Quali estensioni sarebbero possibili?
        Una prima estensione è aggiungere un compartimento E di Esposti,
        ottenendo un modello SEIR. Questo aggiunge uno stato in più e
        arricchisce la dinamica, ma la struttura markoviana rimane invariata.
        Una seconda estensione è rendere β e γ dipendenti dal tempo, per
        simulare l'effetto di misure di contenimento che riducono i contatti.
        Una terza estensione è confrontare la simulazione stocastica con
        la soluzione delle equazioni differenziali ordinarie SIR nel
        limite di popolazione grande.
        Tutte queste estensioni mantengono la struttura di catena di Markov.
    }
\end{frame}

\begin{frame}{Conclusioni}
    \begin{itemize}
        \item Modello SIR = \textbf{catena di Markov a tempo discreto}.
        \item Stati \textbf{assorbenti} ($I=0$) e \textbf{transitori} ($I>0$).
        \item Simulazione Monte Carlo: stima di $\tau$, picco, $R_\infty$.
        \item \textbf{Conferma teorica}: assorbimento certo, variabilità stocastica.
        \item Applicazione concreta del \textbf{formalismo markoviano} (Cap. 7).
    \end{itemize}
