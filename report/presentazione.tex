% SLIDE 2 — Introduzione
\begin{frame}{Introduzione e Motivazione}
    \begin{columns}[c]
        \begin{column}{0.54\textwidth}
            \begin{itemize}
                \item \textbf{Modello SIR}: descrive la dinamica di un'infezione in una popolazione chiusa di $N$ individui.
                \vspace{0.1cm}
                \item \textbf{Vettore di Stato}: a ogni giorno $t \in \mathbb{N}_0$, la popolazione è partizionata in tre gruppi:
                \[ X_t = (S_t, I_t, R_t) \in \mathbb{N}_0^3 \]
                \item \textbf{Conservazione Globale}: $S_t + I_t + R_t = N \implies R_t = N - S_t - I_t$.
                \vspace{0.1cm}
                \item \textbf{Proprietà di Markov}: il futuro dipende unicamente dallo stato presente, rendendo il sistema una catena di Markov a stati finiti.
            \end{itemize}
        \end{column}
        \begin{column}{0.44\textwidth}
            \begin{center}
                \includegraphics[width=\textwidth]{../img/sir_compartments_concept.jpg}
                \vspace{0.05cm}
                {\tiny \textit{Dinamica a 3 compartimenti: $S$ (blu), $I$ (rosso), $R$ (verde)}}
            \end{center}
        \end{column}
    \end{columns}
    
    \note{
        Il modello SIR descrive l'evoluzione di una malattia infettiva in una popolazione chiusa di N individui.
        In ogni istante t, la popolazione è divisa in tre compartimenti: Suscettibili S, Infetti I e Rimossi R.
        Poiché la somma è costante e pari a N, il sistema è bidimensionale.
        La dinamica gode dell'assenza di memoria: l'evoluzione futura dipende unicamente dal numero corrente di persone sane e infette.
    }
\end{frame}

% SLIDE 3 — Contesto Storico ed Epistemologico
\begin{frame}{Contesto Storico: Dal Determinismo alla Stocasticità}
    \begin{columns}[T]
        \begin{column}{0.49\textwidth}
            \begin{block}{\footnotesize 1927 — Kermack \& McKendrick (ODE)}
                \footnotesize \textbf{Dinamica Continua Deterministica}:
                \[
                \begin{cases}
                    \frac{ds}{dt} = -\beta s i \\
                    \frac{di}{dt} = \beta s i - \gamma i \\
                    \frac{dr}{dt} = \gamma i
                \end{cases}
                \]
                \begin{itemize}
                    \item Assume popolazione infinita e densità continue.
                    \item Traiettoria unica senza variabilità: non prevede l'estinzione precoce casuale.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.49\textwidth}
            \begin{block}{\footnotesize 1949 — Bartlett (Markov Discreto)}
                \footnotesize \textbf{Dinamica Stocastica su Popolazione Finita}:
                \[
                (S_{t+1}, I_{t+1}, R_{t+1}) = f(S_t, I_t, \text{eventi casuali})
                \]
                \begin{itemize}
                    \item Individui discreti con contagi e guarigioni binomiali.
                    \item Cattura la varianza tra epidemie e l'assorbimento finale $\mathbb{P}(\tau < \infty)=1$.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

    \vspace{0.2cm}
    \begin{center}
        \footnotesize \textbf{Limite Fluido}: per $N \to \infty$, la media stocastica converge alla traiettoria continua ODE.
    \end{center}

    \note{
        Nel 1927 Kermack e McKendrick proposero il modello continuo deterministico ad equazioni differenziali ordinarie.
        Questo modello funziona molto bene come approssimazione asintotica quando la popolazione è grandissima, ma non può catturare le fluttuazioni casuali a popolazione finita.
        Nel 1949 Bartlett e i probabilisti moderni introdussero il punto di vista stocastico markoviano, dove ogni contagio e guarigione è un evento aleatorio discreto.
    }
\end{frame}

% SLIDE 5 — Richiamo: Catene di Markov
\begin{frame}{Richiamo Teorico: Catene di Markov a Tempo Discreto}
    \begin{block}{Definizione (Proprietà di Markov)}
        Un processo stocastico $\{X_t\}_{t \ge 0}$ su uno spazio di stati discreto $E$ è una \textbf{catena di Markov} se:
        \[ \mathbb{P}(X_{t+1} = j \mid X_t = i, X_{t-1} = i_{t-1}, \dots, X_0 = i_0) = \mathbb{P}(X_{t+1} = j \mid X_t = i) = P(i,j) \]
    \end{block}

    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Matrice di Transizione $P$}:
            \begin{itemize}
                \item $P(i,j) \ge 0$ per ogni coppia $i, j \in E$.
                \item $P$ è \textbf{stocastica per righe}: $\sum_{j \in E} P(i,j) = 1$.
                \item Transizioni a $n$ passi: $P^{(n)} = P^n$.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Stati Assorbenti e Transitori}:
            \begin{itemize}
                \item Uno stato $i$ è \textbf{assorbente} se $P(i,i) = 1$.
                \item Uno stato è \textbf{transitorio} se il processo può uscirne senza più potervi tornare.
            \end{itemize}
        \end{column}
    \end{columns}

    \note{
        Richiamiamo le nozioni fondamentali della teoria markoviana.
        La proprietà di Markov stabilisce l'assenza di memoria: il futuro dipende solo dallo stato corrente.
        La matrice P racchiude le probabilità di transizione a un passo ed è stocastica.
        Un concetto cardine per il nostro modello è la distinzione tra stati transitori (infezione attiva) e stati assorbenti (infezione estinta).
    }
\end{frame}

        \draw[transarrow] (S) -- node[above, font=\footnotesize\bfseries, color=black] {$C_t \sim \text{Bin}(S_t, \frac{\beta I_t}{N})$} (I);
        \draw[transarrow] (I) -- node[above, font=\footnotesize\bfseries, color=black] {$G_t \sim \text{Bin}(I_t, \gamma)$} (R);
    \end{tikzpicture}
    \end{center}

    \vspace{0.15cm}
    \begin{center}
    \resizebox{0.92\textwidth}{!}{
    \begin{tabular}{lll}
    \toprule
    \textbf{Compartimento} & \textbf{Descrizione} & \textbf{Proprietà nella Catena} \\
    \midrule
    $S_t$ (Suscettibili) & Individui sani che possono contrarre il virus & Decresce monotonicamente ($S_{t+1} \le S_t$) \\
    $I_t$ (Infetti) & Individui contagiosi che trasmettono la malattia & Regola l'arresto ($I_t = 0 \implies$ stato assorbente) \\
    $R_t$ (Rimossi) & Individui guariti e immunizzati in modo permanente & Cresce monotonicamente ($R_{t+1} \ge R_t$) \\
    \bottomrule
    \end{tabular}
    }
    \end{center}

    \note{
        Ecco lo schema a blocchi del modello SIR.
        In ogni passo temporale avvengono due processi casuali indipendenti:
        i contagi C_t estratti da una Binomiale con parametro di successo beta*I/N, e le guarigioni G_t con parametro gamma.
        La crescita monotona di R_t garantisce che la catena non abbia cicli.
    }
\end{frame}

% SLIDE 7 — Spazio degli stati
\begin{frame}{Spazio degli Stati $E$}
    \textbf{Definizione Formale}:
    \[ E = \bigl\{(s, i, r) \in \mathbb{N}_0^3 : s + i + r = N\bigr\} \]

    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Cardinalità combinatoria}:
            \[ |E| = \binom{N+2}{2} = \frac{(N+1)(N+2)}{2} \]
            
            \textbf{Esempi numerici}:
            \begin{itemize}
                \item $N = 3 \implies |E| = \mathbf{10}$ stati.
                \item $N = 100 \implies |E| = \mathbf{5151}$ stati.
            \end{itemize}
        \end{column}
        \begin{column}{0.50\textwidth}
            \begin{block}{Partizione dello Spazio $E$}
                \begin{itemize}
                    \item \textbf{Stati Assorbenti $\mathcal{A}$} ($I=0$):
                    \[ \mathcal{A} = \{(s, 0, r) \in E\}, \; |\mathcal{A}| = N+1 \]
                    \item \textbf{Stati Transitori $\mathcal{T}$} ($I>0$):
                    \[ \mathcal{T} = \{(s, i, r) : i>0\}, \; |\mathcal{T}| = \frac{N(N+1)}{2} \]
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

    \note{
        Lo spazio degli stati è l'insieme delle terne di interi non negativi a somma N.
        La formula combinatoria dà (N+1)(N+2)/2 stati.
        Lo spazio si divide esattamente in N+1 stati assorbenti (quelli con I=0) e N(N+1)/2 stati transitori (con I>0).
    }
\end{frame}

% SLIDE 8 — Probabilità di transizione
\begin{frame}{Probabilità di Transizione ed Equazioni di Aggiornamento}
    Per passare dallo stato attuale $(S_t, I_t)$ allo stato futuro $(S_{t+1}, I_{t+1})$, operano due variabili binomiali:

    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{1. Nuovi Contagi ($S \to I$)}
                Probabilità di contagio per suscettibile: $p_{\mathrm{SI}} = \beta \frac{I_t}{N}$.
                \[ C_t \sim \text{Bin}\left(S_t,\, \beta \frac{I_t}{N}\right) \]
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{2. Nuove Guarigioni ($I \to R$)}
                Probabilità di guarigione per infetto: $\gamma \in [0,1]$.
                \[ G_t \sim \text{Bin}(I_t,\, \gamma) \]
            \end{block}
        \end{column}
    \end{columns}

    \vspace{0.15cm}
    \textbf{Probabilità Congiunta di Transizione (Indipendenza Condizionale)}:
    \footnotesize
    \[
    \mathbb{P}(C_t = c, \, G_t = g \mid S_t, I_t) = \underbrace{\mathbb{P}(C_t = c \mid S_t, I_t)}_{\text{Marginale Contagi}} \cdot \underbrace{\mathbb{P}(G_t = g \mid I_t)}_{\text{Marginale Guarigioni}}
    \]
    \normalsize

    \textbf{Aggiornamento dello stato}: $S_{t+1} = S_t - C_t$, \quad $I_{t+1} = I_t + C_t - G_t$, \quad $R_{t+1} = R_t + G_t$.

    \note{
        Per passare dallo stato attuale a quello futuro calcoliamo la probabilità congiunta di avere simultaneamente c contagi e g guarigioni.
        Grazie all'indipendenza condizionale tra contagio e guarigione nello stesso passo temporale, la probabilità congiunta è il prodotto esatto delle due probabilità marginali binomiali.
    }
\end{frame}

% SLIDE 9 — Esempio N=3
\begin{frame}{Esempio Analitico: $N=3$, $\beta=0.5$, $\gamma=0.3$}
    \textbf{Scopo}: Calcolare gli elementi della matrice $P\bigl((s,i,r), (s',i',r')\bigr) = \mathbb{P}(C=c, G=g)$.

    Nello stato \textbf{$(S=2, I=1, R=0)$}, le transizioni possibili sono $(S+1) \times (I+1) = 3 \times 2 = \mathbf{6}$:
    \[
    \mathbb{P}(C=c, G=g) = \binom{2}{c} \left(\tfrac{1}{6}\right)^c \left(\tfrac{5}{6}\right)^{2-c} \cdot \binom{1}{g} (0.3)^g (0.7)^{1-g}
    \]
    \footnotesize \textbf{Esempio di calcolo}: $\mathbb{P}(C=0, G=1) = \left(\tfrac{5}{6}\right)^2 \cdot 0.3 = \mathbf{0.2083} \implies P\bigl((2,1,0), (2,0,1)\bigr) = 0.2083$
    \normalsize

    \vspace{-0.15cm}
    \begin{center}
    \resizebox{0.92\textwidth}{!}{
    \begin{tabular}{ccccc}
    \toprule
    $C$ (contagi) & $G$ (guarigioni) & Stato arrivo $(s',i',r')$ & Formula analitica & Probabilità $P(x, x')$ \\
    \midrule
    0 & 0 & $(2, 1, 0)$ & $(5/6)^2 \cdot 0.7$ & $\mathbf{0.4861}$ \\
    0 & 1 & $(2, 0, 1)$ \textsuperscript{(*)} & $(5/6)^2 \cdot 0.3$ & $\mathbf{0.2083}$ \\
    1 & 0 & $(1, 2, 0)$ & $2(1/6)(5/6) \cdot 0.7$ & $\mathbf{0.1944}$ \\
    1 & 1 & $(1, 1, 1)$ & $2(1/6)(5/6) \cdot 0.3$ & $\mathbf{0.0833}$ \\
    2 & 0 & $(0, 3, 0)$ & $(1/6)^2 \cdot 0.7$ & $\mathbf{0.0194}$ \\
    2 & 1 & $(0, 2, 1)$ & $(1/6)^2 \cdot 0.3$ & $\mathbf{0.0083}$ \\
    \bottomrule
    \end{tabular}
    }
    \end{center}
    \vspace{-0.05cm}
    \footnotesize \textbf{Verifica stocasticità riga}: $\sum P((2,1,0), \cdot) = 0.4861 + 0.2083 + 0.1944 + 0.0833 + 0.0194 + 0.0083 = \mathbf{1.0000}$ \checkmark

    \note{
        Questa formula congiunta serve a generare le singole entrate della matrice di transizione P.
        Per ogni coppia (c,g), il risultato numerico è l'esatta probabilità di transizione tra lo stato presente e quello futuro.
        Notiamo l'esempio esplicito: P((2,1,0),(2,0,1)) = 0.2083, che porta direttamente allo stato assorbente.
    }
\end{frame}

% SLIDE 10 — Matrice di transizione P
\begin{frame}{Matrice di Transizione $P$ e Forma Canonica}
    \textbf{Formula di Transizione per Elemento}:
    \[
    P\bigl((s,i,r),\, (s',i',r')\bigr) = \underbrace{\mathbb{P}(C = s - s')}_{\text{Nuovi Contagi } c} \;\times\; \underbrace{\mathbb{P}(G = r' - r)}_{\text{Nuove Guarigioni } g}
    \]
    \footnotesize con vincolo di bilancio $i' = i + (s - s') - (r' - r)$ (altrimenti $P = 0$).
    \normalsize

    \vspace{0.15cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Proprietà della Matrice $P$}
                \begin{itemize}
                    \item $P$ è \textbf{stocastica per righe}: $\sum_y P(x,y) = 1$.
                    \item \textbf{Triangolare a blocchi}: i rimossi $R_t$ non possono diminuire (assenza di cicli).
                    \item Gli stati con $i=0$ sono assorbenti ($P_{ii}=1$).
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Forma Canonica Assorbente}
                Ordinando prima i transitori $\mathcal{T}$ e poi gli assorbenti $\mathcal{A}$:
                \[ P = \begin{pmatrix} Q & R \\ \mathbf{0} & I \end{pmatrix} \]
                \begin{itemize}
                    \item $Q$: transizioni interne tra stati con virus attivo ($\mathcal{T} \to \mathcal{T}$).
                    \item $R$: transizioni verso l'estinzione ($\mathcal{T} \to \mathcal{A}$).
                    \item $I$: stati assorbenti con virus estinto.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

    \note{
        La probabilità di transizione tra due stati si calcola direttamente identificando il numero di contagi c = s - s' e di guarigioni g = r' - r.
        La matrice P viene riordinata nella forma canonica standard per catene assorbenti con i 4 blocchi Q, R, 0, I.
    }
\end{frame}

% SLIDE 12 — Classificazione degli Stati
\begin{frame}{Classificazione degli Stati e Teorema di Assorbimento}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{Teorema (Assorbimento Quasi Certo)}
                Partendo da qualsiasi stato con infetti $I_0 > 0$, l'estinzione avviene in tempo finito con probabilità 1:
                \[ \mathbb{P}(\tau < \infty) = 1, \quad \tau = \inf\{t \ge 0 : I_t = 0\} \]
                \footnotesize \textbf{Dim.}: da ogni stato in $\mathcal{T}$ si raggiunge $\mathcal{A}$ con probabilità $\ge \gamma^I > 0$ (sole guarigioni).
            \end{alertblock}
            \vspace{0.05cm}
            \begin{block}{\footnotesize Matrice Fondamentale}
                \[ N_{\text{fund}} = (I - Q)^{-1} = \sum_{k=0}^\infty Q^k \]
                \footnotesize Numero medio di visite in ogni stato transitorio prima dell'estinzione.
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Tempo Medio di Assorbimento}
                Per First-Step Analysis, il vettore dei tempi medi $\mathbf{t} = (\mathbb{E}_i[\tau])_{i \in \mathcal{T}}$ soddisfa:
                \[ \mathbf{t} = \mathbf{1} + Q \mathbf{t} \implies (I - Q)\mathbf{t} = \mathbf{1} \]
                da cui:
                \[ \mathbf{t} = (I - Q)^{-1}\mathbf{1} \]
            \end{block}
            \begin{block}{\footnotesize Probabilità di Assorbimento Finale}
                La matrice $B = (I - Q)^{-1} R$ assegna la probabilità di assorbimento in ciascuno stato $(N-r, 0, r)$.
            \end{block}
        \end{column}
    \end{columns}

    \note{
        La teoria ci dà due risultati chiave:
        1) L'epidemia si estingue certamente in tempo finito (probabilità 1) perché da ogni stato c'è un cammino di sole guarigioni con probabilità positiva.

\begin{frame}[fragile]{Implementazione Python (\texttt{src/model.py})}
\begin{lstlisting}[language=Python]
import numpy as np

def next_state(s: int, i: int, r: int, n: int = 100, 
               beta: float = 0.2, gamma: float = 0.1):
    """Esegue un singolo passo stocastico della Catena di Markov SIR."""
    if i == 0:
        return s, 0, r  # Stato assorbente raggiunto

    p_si = beta * i / n
    new_infections = int(np.random.binomial(s, p_si))
    recoveries     = int(np.random.binomial(i, gamma))

    s_next = s - new_infections
    i_next = i + new_infections - recoveries
    r_next = r + recoveries

    return s_next, i_next, r_next
\end{lstlisting}

    \note{
        Il codice sorgente in Python è compatto ed elegante.
        La funzione next_state incapsula la transizione markoviana a 1 passo.
        Se I=0 la catena è assorbita; altrimenti estrae i contagi e le guarigioni con numpy.random.binomial.
    }
\end{frame}
